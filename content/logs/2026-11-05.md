document.addEventListener("DOMContentLoaded", () => {
    const container = document.getElementById("calendar-container");
    const streamContainer = document.getElementById("log-stream") || document.querySelector(".log-stream");
    const tagCloud = document.getElementById("tag-cloud");
    const tagEmptyState = document.getElementById("tag-empty-state");
    const activeTagStatus = document.getElementById("active-tag-status");
    const activeTagLabel = document.getElementById("active-tag-label");
    const tagResetBtn = document.getElementById("tag-reset-btn");

    if (!container) return;

    let currentDate = new Date();

    const badgeDateEl = document.querySelector(".badge-date");
    if (badgeDateEl && badgeDateEl.textContent.trim().match(/^\d{4}-\d{2}-\d{2}$/)) {
        const [y, m, d] = badgeDateEl.textContent.trim().split("-").map(Number);
        currentDate = new Date(y, m - 1, d);
    }

    const cardCache = new Map();
    let activeMonth = `${currentDate.getFullYear()}-${String(currentDate.getMonth() + 1).padStart(2, '0')}`;

    const normalizeTag = (tag) => String(tag || "").trim();

    function getTagsForDate(dateStr) {
        if (!window.logTagMap || !window.logTagMap[dateStr]) return [];
        const raw = window.logTagMap[dateStr];
        if (Array.isArray(raw)) return raw.map(normalizeTag).filter(Boolean);
        if (typeof raw === "string") return raw.split(",").map(normalizeTag).filter(Boolean);
        return [];
    }

    function getAllTagCounts() {
        const counts = new Map();
        for (const dateStr of Object.keys(window.logTagMap || {})) {
            getTagsForDate(dateStr).forEach((tag) => {
                counts.set(tag, (counts.get(tag) || 0) + 1);
            });
        }
        return [...counts.entries()].sort((a, b) => a[0].localeCompare(b[0]));
    }

    function getCurrentSelectedTag() {
        return window.tagFilterState && window.tagFilterState.selectedTag ? window.tagFilterState.selectedTag : null;
    }

    function matchesSelectedTag(dateStr) {
        const selectedTag = getCurrentSelectedTag();
        if (!selectedTag) return true;
        return getTagsForDate(dateStr).includes(selectedTag);
    }

    function updateTagStatus() {
        const selectedTag = getCurrentSelectedTag();

        if (!tagEmptyState || !activeTagStatus || !activeTagLabel) return;

        if (!selectedTag) {
            activeTagStatus.hidden = true;
            return;
        }

        activeTagStatus.hidden = false;
        activeTagLabel.textContent = `[${selectedTag}]`;
    }

    function renderTagCloud() {
        if (!tagCloud) return;

        const counts = getAllTagCounts();
        const selectedTag = getCurrentSelectedTag();

        if (!counts.length) {
            tagCloud.innerHTML = "";
            tagEmptyState.hidden = false;
            activeTagStatus.hidden = true;
            return;
        }

        tagEmptyState.hidden = true;
        tagCloud.innerHTML = counts.map(([tag, count]) => {
            const activeClass = selectedTag === tag ? "selected" : "";
            return `<button type="button" class="tag-cloud-item ${activeClass}" data-tag="${tag}" aria-pressed="${selectedTag === tag}">${tag}</button>`;
        }).join("");

        tagCloud.querySelectorAll(".tag-cloud-item").forEach((button) => {
            button.addEventListener("click", () => {
                const tag = button.dataset.tag;
                const nextTag = getCurrentSelectedTag() === tag ? null : tag;
                window.tagFilterState.selectedTag = nextTag;
                renderTagCloud();
                applyFilterToVisibleMonthCards();
            });
        });

        updateTagStatus();
    }

    function applyFilterToVisibleMonthCards() {
        if (!streamContainer) return;

        const selectedTag = getCurrentSelectedTag();
        const cards = [...streamContainer.querySelectorAll(".log-card[id^='log-']")];

        let visibleCount = 0;
        cards.forEach((card) => {
            const dateStr = card.id.replace("log-", "");
            const visible = matchesSelectedTag(dateStr);
            card.hidden = !visible;
            card.style.display = visible ? "" : "none";
            if (visible) visibleCount += 1;
        });

        if (selectedTag && visibleCount === 0) {
            streamContainer.innerHTML = `
                <div class="log-card" id="no-logs-msg">
                    <p>No logs match the tag <span class="tag-pill">[${selectedTag}]</span>.</p>
                </div>`;
            return;
        }

        if (!selectedTag) {
            const currentMonthDates = Object.keys(logMap || {})
                .filter((date) => date.startsWith(activeMonth))
                .sort()
                .reverse();

            if (currentMonthDates.length === 0) {
                streamContainer.innerHTML = `
                    <div class="log-card" id="no-logs-msg">
                        <p>No logs found for this month. Click a date on the calendar to view past archives.</p>
                    </div>`;
            }
        }
    }

    if (tagResetBtn) {
        tagResetBtn.addEventListener("click", () => {
            window.tagFilterState.selectedTag = null;
            renderTagCloud();
            applyFilterToVisibleMonthCards();
        });
    }

    document.addEventListener("keydown", (event) => {
        if (event.key === "Escape" && window.tagFilterState && window.tagFilterState.selectedTag) {
            window.tagFilterState.selectedTag = null;
            renderTagCloud();
            applyFilterToVisibleMonthCards();
        }
    });

    async function fetchLogCard(dateStr, url) {
        if (cardCache.has(dateStr)) {
            return cardCache.get(dateStr);
        }

        try {
            const response = await fetch(url);
            if (!response.ok) throw new Error(`HTTP ${response.status}`);

            const htmlText = await response.text();
            const doc = new DOMParser().parseFromString(htmlText, "text/html");

            let card = doc.querySelector(".log-card") || doc.querySelector("article");

            if (!card) {
                const mainContent = doc.querySelector(".content-area") || doc.querySelector("main") || doc.body;
                card = document.createElement("article");
                card.className = "log-card";
                card.innerHTML = `
                    <div class="log-meta"><span class="badge-date">${dateStr}</span></div>
                    <div class="log-body">${mainContent.innerHTML}</div>
                `;
            }

            card.id = `log-${dateStr}`;
            const cardHTML = card.outerHTML;
            cardCache.set(dateStr, cardHTML);
            return cardHTML;
        } catch (err) {
            console.error(`Failed to fetch log for ${dateStr} at ${url}:`, err);
            return null;
        }
    }

    async function displayMonthLogs(year, month, targetDateStr = null) {
        if (!streamContainer) return;

        const formattedMonth = String(month + 1).padStart(2, '0');
        const monthPrefix = `${year}-${formattedMonth}`;
        activeMonth = monthPrefix;

        const matchingDates = typeof logMap !== 'undefined'
            ? Object.keys(logMap).filter(date => date.startsWith(monthPrefix)).sort().reverse()
            : [];

        if (matchingDates.length === 0) {
            streamContainer.innerHTML = `
                <div class="log-card" id="no-logs-msg">
                    <p>No logs found for this month. Click a date on the calendar to view past archives.</p>
                </div>`;
            return;
        }

        const filteredDates = matchingDates.filter(date => matchesSelectedTag(date));
        const datesToRender = filteredDates.length > 0 || !getCurrentSelectedTag() ? matchingDates : filteredDates;

        const uncached = datesToRender.filter(d => !cardCache.has(d));
        if (uncached.length > 0) {
            streamContainer.style.opacity = "0.5";
        }

        const cardHTMLs = await Promise.all(
            datesToRender.map(dateStr => fetchLogCard(dateStr, logMap[dateStr]))
        );

        if (activeMonth !== monthPrefix) return;

        streamContainer.style.opacity = "1";
        const validCards = cardHTMLs.filter(Boolean);

        if (validCards.length === 0) {
            streamContainer.innerHTML = `
                <div class="log-card" id="no-logs-msg">
                    <p>No logs match the selected tag for ${monthPrefix}.</p>
                </div>`;
            return;
        }

        streamContainer.innerHTML = validCards.join("");

        if (typeof window.renderMermaid === "function") {
            window.renderMermaid();
        }

        if (targetDateStr) {
            const targetElement = document.getElementById(`log-${targetDateStr}`);
            if (targetElement) {
                targetElement.scrollIntoView({ behavior: "smooth", block: "start" });
                targetElement.classList.add("highlight-log");
                setTimeout(() => targetElement.classList.remove("highlight-log"), 2000);
            }
        }

        applyFilterToVisibleMonthCards();
    }

    function renderCalendar(year, month) {
        container.innerHTML = "";

        const header = document.createElement("div");
        header.className = "calendar-header";

        const prevBtn = document.createElement("button");
        prevBtn.textContent = "←";
        prevBtn.onclick = () => {
            currentDate.setMonth(currentDate.getMonth() - 1);
            update();
        };

        const title = document.createElement("span");
        const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
        title.textContent = `${monthNames[month]} ${year}`;

        const nextBtn = document.createElement("button");
        nextBtn.textContent = "→";
        nextBtn.onclick = () => {
            currentDate.setMonth(currentDate.getMonth() + 1);
            update();
        };

        header.append(prevBtn, title, nextBtn);
        container.appendChild(header);

        const grid = document.createElement("div");
        grid.className = "calendar-grid";

        ["Su", "Mo", "Tu", "We", "Th", "Fr", "Sa"].forEach(day => {
            const dayHeader = document.createElement("div");
            dayHeader.className = "day-header";
            dayHeader.textContent = day;
            grid.appendChild(dayHeader);
        });

        const firstDayIndex = new Date(year, month, 1).getDay();
        const totalDays = new Date(year, month + 1, 0).getDate();

        for (let i = 0; i < firstDayIndex; i++) {
            const emptyCell = document.createElement("div");
            emptyCell.className = "calendar-cell empty";
            grid.appendChild(emptyCell);
        }

        for (let day = 1; day <= totalDays; day++) {
            const cell = document.createElement("div");
            cell.className = "calendar-cell";
            cell.textContent = day;

            const formattedMonth = String(month + 1).padStart(2, '0');
            const formattedDay = String(day).padStart(2, '0');
            const dateStr = `${year}-${formattedMonth}-${formattedDay}`;

            if (typeof logMap !== 'undefined' && logMap[dateStr]) {
                cell.classList.add("has-log");
                cell.title = `View log for ${dateStr}`;
                cell.onclick = () => {
                    if (streamContainer) {
                        displayMonthLogs(year, month, dateStr);
                    } else {
                        window.location.href = logMap[dateStr];
                    }
                };
            } else {
                cell.classList.add("no-log");
            }

            grid.appendChild(cell);
        }

        container.appendChild(grid);
    }

    function update() {
        const yr = currentDate.getFullYear();
        const mo = currentDate.getMonth();
        renderCalendar(yr, mo);
        displayMonthLogs(yr, mo);
    }

    renderTagCloud();
    renderCalendar(currentDate.getFullYear(), currentDate.getMonth());
    displayMonthLogs(currentDate.getFullYear(), currentDate.getMonth());
});
