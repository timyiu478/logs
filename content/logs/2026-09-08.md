#### Rust: Closures

Caching: an example that the `sort_by_key` requires a closure that have `FnMut` trait but not `Fn` trait

```rust
use std::collections::HashMap;

// Simulates an expensive operation (e.g., DB lookup, complex regex, heavy
math)
fn compute_expensive_score(word: &str) -> usize {
    word.chars().filter(|c| "aeiou".contains(*c)).count()
}

fn main() {
    let mut words = vec!["banana", "apple", "cherry", "banana", "apple",
"date"];

    let mut cache = HashMap::new();

    // Requires `FnMut` because `cache` is mutated on cache misses
    words.sort_by_key(|&word| {
        *cache.entry(word).or_insert_with(|| compute_expensive_score(word))
    });

    println!("{words:?}");
    println!("Cached {} unique keys.", cache.len());
}
```
